Google 試算表 (Google Sheets) 自動分班連動

自動分工作表 (Tabs) 功能說明：

下方 Apps Script 程式碼已升級！會自動依據送出的班級名稱（如「高二班級」、「高三班級」），在您的 Google 試算表中自動建立並記錄至對應的工作表頁籤，方便您分別整理高二與高三的點名資料。

  1. 開啟 Google 試算表，點擊選單 擴充功能 ➔ Apps Script。
  2. 刪除原本程式碼，將下方【自動分班版專用程式碼】貼上並存檔。
  3. 點擊右上角 部署 ➔ 新建部署（類型選「Web 應用程式」，存取權限設為「任何人」），將複製的網址貼在下方輸入框。
Google Apps Script 自動分班專用程式碼： 
function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var data = JSON.parse(e.postData.contents);
  var className = data.className || "全班點名";
  
  // 自動以班級名稱尋找或新建工作表 (Sheet Tab)
  var sheet = ss.getSheetByName(className);
  if (!sheet) {
    sheet = ss.insertSheet(className);
  }
  
  // 若工作表為全新空白，自動新增欄位標頭
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["班級", "點名日期", "總人數", "出席人數", "缺席人數", "請假人數", "出席率", "缺席學生名單", "請假學生名單", "紀錄時間"]);
  }
  
  sheet.appendRow([
    className,
    data.date,
    data.total,
    data.present,
    data.absent,
    data.excused,
    data.rate + "%",
    data.absentList.join(", "),
    data.excusedList.join(", "),
    new Date().toLocaleString("zh-TW", { timeZone: "Asia/Taipei" })
  ]);
  
  return ContentService.createTextOutput(JSON.stringify({ status: "success" }))
    .setMimeType(ContentService.MimeType.JSON);
}
