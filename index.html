「學生資料」試算表雙向連動與同步設定

「學生資料」試算表連動說明：

已完美適應您的「學生資料」試算表（包含 `高二班級` 與 `高三班級` 頁籤）。透過下方 Apps Script，網頁可以直接讀取這兩個頁籤裡的名單，並將點名紀錄自動記錄於對應的分頁中！

  1. 在您的「學生資料」試算表中，點擊選單 擴充功能 ➔ Apps Script。
  2. 將裡面的程式碼全部刪除，貼上下方專用程式碼並存檔。
  3. 點擊右上角 部署 ➔ 新建部署（類型選「Web 應用程式」，存取權限設為「任何人」），將網址貼在最下方！
Apps Script 雙向同步專用程式碼： 
function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var action = e.parameter.action;
  
  if (action === "getRoster") {
    var className = e.parameter.className || "高二班級";
    var sheet = ss.getSheetByName(className) || ss.getSheets()[0];
    var data = sheet.getDataRange().getValues();
    var roster = [];
    
    for (var i = 1; i < data.length; i++) {
      var row = data[i];
      if (row[0] || row[1] || row[2]) {
        roster.push({
          homeClass: row[0] ? String(row[0]) : "",
          studentId: row[1] ? String(row[1]) : "",
          name: row[2] ? String(row[2]) : (row[1] ? String(row[1]) : String(row[0]))
        });
      }
    }
    
    return ContentService.createTextOutput(JSON.stringify({ status: "success", className: className, roster: roster }))
      .setMimeType(ContentService.MimeType.JSON);
  }
  
  return ContentService.createTextOutput("Google Sheets Webhook Ready");
}

function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var data = JSON.parse(e.postData.contents);
  var className = data.className || "高二班級";
  
  var sheet = ss.getSheetByName(className);
  if (!sheet) {
    sheet = ss.insertSheet(className);
  }
  
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["課程班級", "點名日期", "總人數", "出席人數", "缺席人數", "請假人數", "出席率", "缺席學生名單(含原班級)", "請假學生名單(含原班級)", "紀錄時間"]);
  }
  
  sheet.appendRow([
    className,
    data.date,
    data.total,
    data.present,
    data.absent,
    data.excused,
    data.rate + "%",
    data.absentList.join(", "),
    data.excusedList.join(", "),
    new Date().toLocaleString("zh-TW", { timeZone: "Asia/Taipei" })
  ]);
  
  return ContentService.createTextOutput(JSON.stringify({ status: "success" }))
    .setMimeType(ContentService.MimeType.JSON);
}
